set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
  add_link_options(-fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createStrelka2 -sINVOKE_RUN=0 -sEXPORTED_RUNTIME_METHODS=FS,callMain -sINCOMING_MODULE_JS_API=preRun,noInitialRun,stdin,stdout,stderr,onExit,quit,locateFile,wasmBinary,print,printErr,onAbort,thisProgram -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sABORTING_MALLOC=1 -sINITIAL_MEMORY=67108864 -sMAXIMUM_MEMORY=2147483648 -sSTACK_SIZE=8388608 -sEXIT_RUNTIME=1 -sFORCE_FILESYSTEM=1 -Wl,--fatal-warnings)
  set(CMAKE_EXECUTABLE_SUFFIX ".mjs")

set(BOOST "${DEPS}/boost_1_58_0_subset")
set(HTS "${DEPS}/htslib-1.7-6-g6d2bfb7")
include_directories("${BOOST}" "${HTS}" "${ZPREFIX}/include" "${DEPS}/CodeMin-1.0.5/include" "${DEPS}/rapidjson-1.1.0/include" "${STRELKA}/src/c++/lib" "${CMAKE_CURRENT_BINARY_DIR}/config")
add_definitions(-DBOOST_ALL_NO_LIB -DBOOST_FILESYSTEM_STATIC_LINK -DBOOST_SYSTEM_STATIC_LINK)
foreach(component date_time filesystem program_options serialization system timer chrono)
  if(component STREQUAL "serialization")
    set(names basic_archive basic_iarchive basic_iserializer basic_oarchive basic_oserializer basic_pointer_iserializer basic_pointer_oserializer basic_serializer_map basic_text_iprimitive basic_text_oprimitive basic_xml_archive binary_iarchive binary_oarchive extended_type_info extended_type_info_typeid extended_type_info_no_rtti polymorphic_iarchive polymorphic_oarchive stl_port text_iarchive text_oarchive void_cast archive_exception xml_grammar xml_iarchive xml_oarchive xml_archive_exception)
    set(units "")
    foreach(name IN LISTS names)
      list(APPEND units "${BOOST}/libs/${component}/src/${name}.cpp")
    endforeach()
  elseif(component STREQUAL "date_time")
    set(units "${BOOST}/libs/date_time/src/gregorian/greg_month.cpp" "${BOOST}/libs/date_time/src/gregorian/greg_weekday.cpp" "${BOOST}/libs/date_time/src/gregorian/date_generators.cpp")
  else()
    file(GLOB units "${BOOST}/libs/${component}/src/*.cpp")
  endif()
  add_library(boost_${component} STATIC ${units})
  if(component STREQUAL "date_time")
    target_compile_definitions(boost_${component} PRIVATE DATE_TIME_INLINE BOOST_DATE_TIME_STATIC_LINK)
  endif()
  if(component STREQUAL "chrono")
    target_compile_definitions(boost_${component} PRIVATE BOOST_SYSTEM_NO_DEPRECATED BOOST_CHRONO_STATIC_LINK)
  endif()
endforeach()

endforeach()

 list(APPEND libraries strelka_${name})
endforeach()

  set(app strelka)
